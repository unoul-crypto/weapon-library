cmake_minimum_required(VERSION 3.20)
project(weapon_library VERSION 0.10.0 LANGUAGES CXX)

option(WEAPON_LIBRARY_BUILD_TESTS "Build headless behavioral tests" OFF)
option(WEAPON_LIBRARY_BUILD_EXAMPLES "Build the headless example" OFF)
option(WEAPON_LIBRARY_BUILD_DEMO "Build the interactive raylib firing range" OFF)
option(WEAPON_LIBRARY_BUILD_BALLISTICS "Build the independent ballistics module" ON)

add_library(weapon_library src/Weapon.cpp src/FiringMechanism.cpp src/WeaponHandling.cpp src/WeaponPresentation.cpp src/MeleeMechanism.cpp)
add_library(weapon_library::weapon_library ALIAS weapon_library)
target_include_directories(weapon_library PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_compile_features(weapon_library PUBLIC cxx_std_17)
set_target_properties(weapon_library PROPERTIES CXX_EXTENSIONS OFF)

if(MSVC)
    target_compile_options(weapon_library PRIVATE /W4 /permissive-)
else()
    target_compile_options(weapon_library PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(WEAPON_LIBRARY_BUILD_BALLISTICS)
    add_library(weapon_library_ballistics src/Ballistics.cpp)
    add_library(weapon_library::ballistics ALIAS weapon_library_ballistics)
    target_include_directories(weapon_library_ballistics PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
    target_compile_features(weapon_library_ballistics PUBLIC cxx_std_17)
    set_target_properties(weapon_library_ballistics PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(weapon_library_ballistics PRIVATE /W4 /permissive-)
    else()
        target_compile_options(weapon_library_ballistics PRIVATE -Wall -Wextra -Wpedantic)
    endif()
elseif(WEAPON_LIBRARY_BUILD_DEMO)
    message(FATAL_ERROR "The firing range requires WEAPON_LIBRARY_BUILD_BALLISTICS=ON")
endif()

if(WEAPON_LIBRARY_BUILD_BALLISTICS AND (WEAPON_LIBRARY_BUILD_DEMO OR WEAPON_LIBRARY_BUILD_TESTS))
    add_library(weapon_range_core examples/range/RangeSimulation.cpp examples/range/RangeVisuals.cpp)
    target_include_directories(weapon_range_core PUBLIC examples/range)
    target_link_libraries(weapon_range_core PUBLIC weapon_library::weapon_library weapon_library::ballistics)
endif()

if(WEAPON_LIBRARY_BUILD_TESTS)
    enable_testing()
    add_executable(weapon_library_tests tests/WeaponTests.cpp)
    target_link_libraries(weapon_library_tests PRIVATE weapon_library::weapon_library)
    add_test(NAME weapon_library.behavior COMMAND weapon_library_tests)
    add_executable(weapon_library_firing_tests tests/FiringTests.cpp)
    target_link_libraries(weapon_library_firing_tests PRIVATE weapon_library::weapon_library)
    add_test(NAME weapon_library.firing COMMAND weapon_library_firing_tests)
    add_executable(weapon_library_reload_tests tests/ReloadTests.cpp)
    target_link_libraries(weapon_library_reload_tests PRIVATE weapon_library::weapon_library)
    add_test(NAME weapon_library.reload COMMAND weapon_library_reload_tests)
    add_executable(weapon_library_handling_tests tests/HandlingTests.cpp)
    target_link_libraries(weapon_library_handling_tests PRIVATE weapon_library::weapon_library)
    add_test(NAME weapon_library.handling COMMAND weapon_library_handling_tests)
    add_executable(weapon_library_presentation_tests tests/PresentationTests.cpp)
    target_link_libraries(weapon_library_presentation_tests PRIVATE weapon_library::weapon_library)
    add_test(NAME weapon_library.presentation COMMAND weapon_library_presentation_tests)
    add_executable(weapon_library_melee_tests tests/MeleeTests.cpp)
    target_link_libraries(weapon_library_melee_tests PRIVATE weapon_library::weapon_library)
    add_test(NAME weapon_library.melee COMMAND weapon_library_melee_tests)
    if(WEAPON_LIBRARY_BUILD_BALLISTICS)
        add_executable(weapon_library_ballistics_tests tests/BallisticsTests.cpp)
        target_link_libraries(weapon_library_ballistics_tests PRIVATE weapon_library::ballistics)
        add_test(NAME weapon_library.ballistics COMMAND weapon_library_ballistics_tests)
        add_executable(weapon_library_range_tests tests/RangeTests.cpp)
        target_link_libraries(weapon_library_range_tests PRIVATE weapon_range_core)
        add_test(NAME weapon_library.range COMMAND weapon_library_range_tests)
        add_executable(weapon_library_visuals_tests tests/VisualsTests.cpp)
        target_link_libraries(weapon_library_visuals_tests PRIVATE weapon_range_core)
        add_test(NAME weapon_library.visuals COMMAND weapon_library_visuals_tests "${CMAKE_CURRENT_SOURCE_DIR}/examples/range/visuals")
    endif()
endif()

if(WEAPON_LIBRARY_BUILD_DEMO)
    enable_language(C)
    include(FetchContent)
    if(POLICY CMP0135)
        cmake_policy(SET CMP0135 NEW)
    endif()
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(raylib
        URL https://codeload.github.com/raysan5/raylib/zip/refs/tags/5.5)
    FetchContent_MakeAvailable(raylib)
    add_executable(weapon_library_range
        examples/range/main.cpp examples/range/RangeRenderer.cpp)
    target_link_libraries(weapon_library_range PRIVATE weapon_range_core raylib)
    target_compile_definitions(weapon_library_range PRIVATE WEAPON_RANGE_VERSION="${PROJECT_VERSION}")
    add_custom_target(weapon_range_visuals
        COMMAND ${CMAKE_COMMAND} -E copy_directory
            "${CMAKE_CURRENT_SOURCE_DIR}/examples/range/visuals"
            "$<TARGET_FILE_DIR:weapon_library_range>/visuals"
        COMMENT "Copying range models and visual settings")
    add_dependencies(weapon_library_range weapon_range_visuals)
endif()

if(WEAPON_LIBRARY_BUILD_EXAMPLES)
    add_executable(weapon_library_example examples/Headless.cpp)
    target_link_libraries(weapon_library_example PRIVATE weapon_library::weapon_library)
    add_executable(weapon_library_firing_example examples/Firing.cpp)
    target_link_libraries(weapon_library_firing_example PRIVATE weapon_library::weapon_library)
endif()
